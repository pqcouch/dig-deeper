
1 1 בִּשְׁנַ֤ת שְׁתַּ֨יִם֙ לְדָרְיָ֣וֶשׁ הַמֶּ֔לֶךְ בַּחֹ֨דֶשׁ֙ הַשִּׁשִּׁ֔י בְּיֹ֥ום אֶחָ֖ד לַחֹ֑דֶשׁ הָיָ֨ה דְבַר־יְהוָ֜ה בְּיַד־חַגַּ֣י הַנָּבִ֗יאa‬ אֶל־זְרֻבָּבֶ֤ל בֶּן־שְׁאַלְתִּיאֵל֙ פַּחַ֣ת יְהוּדָ֔ה וְאֶל־יְהֹושֻׁ֧עַ בֶּן־יְהֹוצָדָ֛ק הַכֹּהֵ֥ן הַגָּדֹ֖ול לֵאמֹֽר׃
2 כֹּ֥ה אָמַ֛ר יְהוָ֥ה צְבָאֹ֖ות לֵאמֹ֑ר הָעָ֤ם הַזֶּה֙ אָֽמְר֔וּ לֹ֥א a‬עֶת־בֹּ֛אa‬ עֶת־בֵּ֥ית יְהוָ֖ה לְהִבָּנֹֽות׃  פ  
3 וַֽיְהִי֙ דְּבַר־יְהוָ֔ה בְּיַד־חַגַּ֥י הַנָּבִ֖יא לֵאמֹֽר׃
4 הַעֵ֤ת לָכֶם֙ אַתֶּ֔ם לָשֶׁ֖בֶת בְּבָתֵּיכֶ֣םa‬ סְפוּנִ֑ים וְהַבַּ֥יִת הַזֶּ֖ה חָרֵֽב׃
5 וְעַתָּ֕ה כֹּ֥ה אָמַ֖ר יְהוָ֣ה צְבָאֹ֑ות שִׂ֥ימוּ לְבַבְכֶ֖ם עַל־דַּרְכֵיכֶֽם׃
6 זְרַעְתֶּ֨ם הַרְבֵּ֜ה וְהָבֵ֣א מְעָ֗ט אָכֹ֤ול וְאֵין־לְשָׂבְעָה֙ שָׁתֹ֣ו וְאֵין־לְשָׁכְרָ֔ה לָבֹ֖ושׁ וְאֵין־לְחֹ֣ם לֹ֑ו וְהַ֨מִּשְׂתַּכֵּ֔ר מִשְׂתַּכֵּ֖רa‬ אֶל־צְרֹ֥ור נָקֽוּב׃  פ  
7 כֹּ֥ה אָמַ֖ר יְהוָ֣ה צְבָאֹ֑ות a‬שִׂ֥ימוּ לְבַבְכֶ֖ם עַל־דַּרְכֵיכֶֽםa‬׃
8 עֲל֥וּ הָהָ֛ר וַהֲבֵאתֶ֥ם עֵ֖ץ וּבְנ֣וּ הַבָּ֑יִת וְאֶרְצֶה־בֹּ֥ו וְאֶכָּבְדָ֖ אָמַ֥ר יְהוָֽה׃
9 פָּנֹ֤ה אֶל־הַרְבֵּה֙ וְהִנֵּ֣הa‬ לִמְעָ֔ט וַהֲבֵאתֶ֥ם הַבַּ֖יִת וְנָפַ֣חְתִּי בֹ֑ו יַ֣עַן מֶ֗ה נְאֻם֙ יְהוָ֣ה צְבָאֹ֔ות יַ֗עַן בֵּיתִי֙ אֲשֶׁר־ה֣וּא חָרֵ֔ב וְאַתֶּ֥ם רָצִ֖ים אִ֥ישׁ לְבֵיתֹֽו׃
10 עַל־כֵּ֣ן עֲלֵיכֶ֔םa‬ כָּלְא֥וּ שָׁמַ֖יִםb‬ מִטָּ֑לc‬ וְהָאָ֖רֶץ כָּלְאָ֥ה יְבוּלָֽהּ׃
11 וָאֶקְרָ֨א חֹ֜רֶב עַל־הָאָ֣רֶץ וְעַל־הֶהָרִ֗ים וְעַל־הַדָּגָן֙ וְעַל־הַתִּירֹ֣ושׁ a‬וְעַל־הַיִּצְהָ֔רa‬ וְעַ֛לb‬ אֲשֶׁ֥ר תֹּוצִ֖יא הָאֲדָמָ֑ה וְעַל־הָֽאָדָם֙ וְעַל־הַבְּהֵמָ֔ה וְעַ֖ל כָּל־יְגִ֥יעַ כַּפָּֽיִם׃  ס  
12 וַיִּשְׁמַ֣ע זְרֻבָּבֶ֣ל׀ בֶּֽן־שַׁלְתִּיאֵ֡ל וִיהֹושֻׁ֣עַ בֶּן־יְהֹוצָדָק֩ הַכֹּהֵ֨ן הַגָּדֹ֜ול וְכֹ֣ל׀ שְׁאֵרִ֣ית הָעָ֗ם בְּקֹול֙ יְהוָ֣ה אֱלֹֽהֵיהֶ֔ם וְעַל־דִּבְרֵי֙ חַגַּ֣י הַנָּבִ֔יא כַּאֲשֶׁ֥ר שְׁלָחֹ֖ו יְהוָ֣ה אֱלֹהֵיהֶ֑םa‬ וַיִּֽירְא֥וּ הָעָ֖ם מִפְּנֵ֥י יְהוָֽה׃
13 וַ֠יֹּאמֶר חַגַּ֞י a‬מַלְאַ֧ךְ יְהוָ֛ה בְּמַלְאֲכ֥וּת יְהוָ֖הa‬ לָעָ֣ם לֵאמֹ֑ר אֲנִ֥י אִתְּכֶ֖ם נְאֻם־יְהוָֽה׃
14 וַיָּ֣עַר יְהוָ֡ה אֶת־רוּחַ֩ זְרֻבָּבֶ֨ל בֶּן־שַׁלְתִּיאֵ֜ל פַּחַ֣ת יְהוּדָ֗ה וְאֶת־ר֨וּחַ֙ יְהֹושֻׁ֤עַ בֶּן־יְהֹוצָדָק֙ הַכֹּהֵ֣ן הַגָּדֹ֔ול וְֽאֶת־ר֔וּחַ כֹּ֖ל שְׁאֵרִ֣ית הָעָ֑ם וַיָּבֹ֨אוּ֙ וַיַּעֲשׂ֣וּ מְלָאכָ֔ה בְּבֵית־יְהוָ֥ה צְבָאֹ֖ות אֱלֹהֵיהֶֽם׃  פ  
15 בְּיֹ֨ום עֶשְׂרִ֧ים וְאַרְבָּעָ֛ה לַחֹ֖דֶשׁ בַּשִּׁשִּׁ֑יa‬ בִּשְׁנַ֥ת שְׁתַּ֖יִם לְדָרְיָ֥וֶשׁ הַמֶּֽלֶךְ׃

2 1 בַּשְּׁבִיעִ֕י בְּעֶשְׂרִ֥ים וְאֶחָ֖ד לַחֹ֑דֶשׁ הָיָה֙ דְּבַר־יְהוָ֔ה בְּיַד־a‬חַגַּ֥י הַנָּבִ֖יא לֵאמֹֽר׃
2 אֱמָר־נָ֗א אֶל־זְרֻבָּבֶ֤ל בֶּן־שַׁלְתִּיאֵל֙ פַּחַ֣ת יְהוּדָ֔ה וְאֶל־יְהֹושֻׁ֥עַ בֶּן־יְהֹוצָדָ֖ק הַכֹּהֵ֣ן הַגָּדֹ֑ול וְאֶל־a‬שְׁאֵרִ֥ית הָעָ֖ם לֵאמֹֽר׃
3 מִ֤י בָכֶם֙ הַנִּשְׁאָ֔ר אֲשֶׁ֤ר רָאָה֙ אֶת־הַבַּ֣יִת הַזֶּ֔ה בִּכְבֹודֹ֖ו הָרִאשֹׁ֑ון וּמָ֨ה אַתֶּ֜ם רֹאִ֤ים אֹתֹו֙ עַ֔תָּה הֲלֹ֥וא כָמֹ֛הוּ כְּאַ֖יִן בְּעֵינֵיכֶֽם׃
4 וְעַתָּ֣ה חֲזַ֣ק זְרֻבָּבֶ֣ל׀ a‬נְאֻם־יְהוָ֡הa‬ וַחֲזַ֣ק יְהֹושֻׁ֣עַ b‬בֶּן־יְהֹוצָדָק֩ הַכֹּהֵ֨ן הַגָּדֹ֜ול וַחֲזַ֨ק כָּל־עַ֥ם הָאָ֛רֶץ נְאֻם־יְהוָ֖הb‬ וַֽעֲשׂ֑וּ כִּֽי־אֲנִ֣י אִתְּכֶ֔ם נְאֻ֖ם יְהוָ֥ה צְבָאֹֽותc‬׃
5 a‬ b‬אֶֽת־הַדָּבָ֞רb‬ אֲשֶׁר־כָּרַ֤תִּי אִתְּכֶם֙ בְּצֵאתְכֶ֣ם מִמִּצְרַ֔יִםa‬ וְרוּחִ֖י עֹמֶ֣דֶת בְּתֹוכְכֶ֑ם אַל־תִּירָֽאוּ׃  ס  
6 כִּ֣י כֹ֤ה אָמַר֙ יְהוָ֣ה צְבָאֹ֔ות a‬עֹ֥וד אַחַ֖ת מְעַ֣ט הִ֑יאa‬ וַאֲנִ֗י מַרְעִישׁ֙ אֶת־הַשָּׁמַ֣יִם וְאֶת־הָאָ֔רֶץ וְאֶתb‬־הַיָּ֖ם וְאֶת־הֶחָרָבָֽה׃
7 וְהִרְעַשְׁתִּי֙ אֶת־כָּל־הַגֹּויִ֔ם וּבָ֖אוּ חֶמְדַּ֣תa‬ כָּל־הַגֹּויִ֑ם וּמִלֵּאתִ֞י אֶת־הַבַּ֤יִת הַזֶּה֙ כָּבֹ֔וד אָמַ֖ר יְהוָ֥ה צְבָאֹֽות׃
8 לִ֥י הַכֶּ֖סֶף וְלִ֣י הַזָּהָ֑ב נְאֻ֖ם יְהוָ֥ה צְבָאֹֽות׃
9 גָּדֹ֣ול יִֽהְיֶ֡ה כְּבֹוד֩ הַבַּ֨יִת הַזֶּ֤ה הָאַֽחֲרֹון֙ מִן־הָ֣רִאשֹׁ֔ון אָמַ֖ר יְהוָ֣ה צְבָאֹ֑ות וּבַמָּקֹ֤ום הַזֶּה֙ אֶתֵּ֣ן שָׁלֹ֔ום נְאֻ֖ם יְהוָ֥ה צְבָאֹֽותa‬׃  פ  
10 בְּעֶשְׂרִ֤ים וְאַרְבָּעָה֙ לַתְּשִׁיעִ֔י בִּשְׁנַ֥ת שְׁתַּ֖יִם לְדָרְיָ֑וֶשׁ הָיָה֙ דְּבַר־יְהוָ֔ה אֶל־חַגַּ֥י הַנָּבִ֖יא לֵאמֹֽר׃
11 כֹּ֥ה אָמַ֖ר יְהוָ֣ה צְבָאֹ֑ות שְׁאַל־נָ֧א אֶת־הַכֹּהֲנִ֛ים תֹּורָ֖ה לֵאמֹֽר׃
12 הֵ֣ן׀ יִשָּׂא־אִ֨ישׁ בְּשַׂר־קֹ֜דֶשׁ בִּכְנַ֣ף בִּגְדֹ֗ו וְנָגַ֣ע בִּ֠כְנָפֹו אֶל־הַלֶּ֨חֶם וְאֶל־הַנָּזִ֜יד וְאֶל־הַיַּ֧יִן וְאֶל־שֶׁ֛מֶן וְאֶל־כָּל־מַאֲכָ֖ל הֲיִקְדָּ֑שׁ וַיַּעֲנ֧וּ הַכֹּהֲנִ֛ים וַיֹּאמְר֖וּ לֹֽא׃
13 וַיֹּ֣אמֶר חַגַּ֔י אִם־יִגַּ֧ע טְמֵא־נֶ֛פֶשׁ בְּכָל־אֵ֖לֶּה הֲיִטְמָ֑א וַיַּעֲנ֧וּ הַכֹּהֲנִ֛ים וַיֹּאמְר֖וּ יִטְמָֽא׃
14 וַיַּ֨עַן חַגַּ֜י וַיֹּ֗אמֶר כֵּ֣ן הָֽעָם־הַ֠זֶּה וְכֵן־הַגֹּ֨וי הַזֶּ֤ה לְפָנַי֙ נְאֻם־יְהוָ֔ה וְכֵ֖ן כָּל־מַעֲשֵׂ֣ה יְדֵיהֶ֑ם וַאֲשֶׁ֥ר יַקְרִ֛יבוּ שָׁ֖ם טָמֵ֥א הֽוּאa‬׃
15 a‬וְעַתָּה֙ שִֽׂימוּ־נָ֣א לְבַבְכֶ֔ם מִן־הַיֹּ֥ום הַזֶּ֖ה וָמָ֑עְלָה מִטֶּ֧רֶם שֽׂוּם־אֶ֛בֶן אֶל־אֶ֖בֶן בְּהֵיכַ֥ל יְהוָֽה׃
16 מִֽהְיֹותָ֥םa‬ בָּא֙b‬ אֶל־עֲרֵמַ֣ת עֶשְׂרִ֔ים וְהָיְתָ֖ה עֲשָׂרָ֑ה בָּ֣א אֶל־הַיֶּ֗קֶב לַחְשֹׂף֙c‬ חֲמִשִּׁ֣ים פּוּרָ֔הd‬ וְהָיְתָ֖ה עֶשְׂרִֽים׃
17 a‬הִכֵּ֨יתִי אֶתְכֶ֜ם בַּשִּׁדָּפֹ֤ון וּבַיֵּֽרָקֹון֙ וּבַבָּרָ֔ד אֵ֖ת כָּל־מַעֲשֵׂ֣ה יְדֵיכֶ֑ם b‬וְאֵין־אֶתְכֶ֥םb‬ אֵלַ֖י נְאֻם־יְהוָֽה׃
18 שִׂימוּ־נָ֣א לְבַבְכֶ֔ם מִן־הַיֹּ֥ום הַזֶּ֖ה וָמָ֑עְלָה a‬מִיֹּום֩ עֶשְׂרִ֨ים וְאַרְבָּעָ֜ה לַתְּשִׁיעִ֗יa‬ b‬לְמִן־הַיֹּ֛ום אֲשֶׁר־יֻסַּ֥ד הֵֽיכַל־יְהוָ֖ה שִׂ֥ימוּ לְבַבְכֶֽםb‬׃
19 הַעֹ֤וד הַזֶּ֨רַע֙a‬ בַּמְּגוּרָ֔ה b‬וְעַד־הַגֶּ֨פֶן וְהַתְּאֵנָ֧ה וְהָרִמֹּ֛וןb‬ וְעֵ֥ץ הַזַּ֖יִת לֹ֣א נָשָׂ֑א מִן־הַיֹּ֥ום הַזֶּ֖ה אֲבָרֵֽךְ׃  ס  
20 וַיְהִ֨י דְבַר־יְהוָ֤ה׀ שֵׁנִית֙ אֶל־חַגַּ֔י בְּעֶשְׂרִ֧ים וְאַרְבָּעָ֛ה לַחֹ֖דֶשׁ לֵאמֹֽר׃
21 אֱמֹ֕ר אֶל־זְרֻבָּבֶ֥ל פַּֽחַת־יְהוּדָ֖ה לֵאמֹ֑ר אֲנִ֣י מַרְעִ֔ישׁ אֶת־הַשָּׁמַ֖יִם וְאֶת־הָאָֽרֶץa‬׃
22 וְהָֽפַכְתִּי֙ כִּסֵּ֣א מַמְלָכֹ֔ות וְהִ֨שְׁמַדְתִּ֔י חֹ֖זֶק מַמְלְכֹ֣ותa‬ הַגֹּויִ֑ם וְהָפַכְתִּ֤י מֶרְכָּבָה֙ וְרֹ֣כְבֶ֔יהָ וְיָרְד֤וּ סוּסִים֙ וְרֹ֣כְבֵיהֶ֔ם b‬אִ֖ישׁ בְּחֶ֥רֶב אָחִֽיוb‬׃
23 בַּיֹּ֣ום הַה֣וּא a‬נְאֻם־יְהוָ֣ה צְבָאֹ֡ותa‬ אֶ֠קָּחֲךָ זְרֻבָּבֶ֨ל a‬בֶּן־שְׁאַלְתִּיאֵ֤לa‬ עַבְדִּי֙ a‬נְאֻם־יְהוָ֔הa‬ וְשַׂמְתִּ֖יךָ כַּֽחֹותָ֑ם כִּֽי־בְךָ֣ בָחַ֔רְתִּי נְאֻ֖ם יְהוָ֥ה צְבָאֹֽות׃‬
